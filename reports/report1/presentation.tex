\section{Motywacja i model SEIRD}

\begin{frame}{Motywacja -- dlaczego modelowanie agentowe?}
  \begin{columns}[T]
    \column{0.48\textwidth}
    \textbf{Klasyczne modele ODE (SIR)}
    \begin{itemize}
      \item Jednorodne mieszanie populacji
      \item Brak struktury przestrzennej
      \item Brak heterogeniczności jednostek
    \end{itemize}
    \[
      \frac{dS}{dt} = -\beta \frac{SI}{N},\;\;
      \frac{dI}{dt} = \beta \frac{SI}{N} - \gamma I
    \]

    \column{0.48\textwidth}
    \textbf{Modelowanie agentowe (ABM)}
    \begin{itemize}
      \item Każdy agent: własne atrybuty i reguły
      \item Realistyczna topologia miejska
      \item Emergentne zjawiska (superspreaderzy, ogniska)
      \item Naturalne modelowanie interwencji
    \end{itemize}
  \end{columns}

  \vspace{0.5cm}
  \begin{block}{Cel projektu}
    Trójstopniowa implementacja wieloskalowej symulacji ABM pandemii
    w środowisku miejskim opartym na węzłach POI z modelem SEIRD.
  \end{block}
\end{frame}

\begin{frame}{Model epidemiologiczny SEIRD}
  \[
    S \xrightarrow{\lambda(t)} E \xrightarrow{\sigma} I \xrightarrow{\gamma} R,
    \qquad I \xrightarrow{\mu} D
  \]

  \begin{columns}[T]
    \column{0.55\textwidth}
    \begin{itemize}
      \item \textbf{S} -- Podatny (brak odporności)
      \item \textbf{E} -- Inkubacja (zainfekowany, niezakaźny, mobilny)
      \item \textbf{I} -- Zakaźny (emituje patogen)
      \item \textbf{R} -- Wyleczony (nabyta odporność)
      \item \textbf{D} -- Zgon
    \end{itemize}
    \vspace{0.3cm}
    \textbf{Ładunek wirusowy} -- ciągła zmienna agenta:
    narasta pod koniec fazy E, szczytuje w~I,\\
    maleje przy zdrowieniu. Bezpośrednio skaluje $p_{\text{inf}}$.

    \column{0.42\textwidth}
    \begin{tabular}{ll}
      \toprule
      Parametr & Wartość \\
      \midrule
      Inkubacja & 4 dni \\
      Zakaźność & 8 dni \\
      $p_{\text{death}}$ & 2\% \\
      $p_{\text{mult}}$ & 0,16 \\
      Cel $R_0$ & 2,5--3,5 \\
      \bottomrule
    \end{tabular}
  \end{columns}
\end{frame}

\section{Architektura systemu}

\begin{frame}{Środowisko: Graf węzłów POI}
  \begin{columns}[T]
    \column{0.50\textwidth}
    Graf $G=(V,E)$ -- węzły to fizyczne lokalizacje:

    \begin{itemize}
      \item \textbf{Gospod.\ domowe:} $\lfloor N/4 \rfloor$ węzłów
      \item \textbf{Biura:} $\lfloor N/25 \rfloor$ węzłów
      \item \textbf{Szkoły:} $\lfloor N/30 \rfloor$ węzłów
      \item \textbf{Sklepy:} $\lfloor N/50 \rfloor$ węzłów
      \item \textbf{Parki:} $\lfloor N/100 \rfloor$ węzłów
      \item \textbf{Placówki zdrowia:} $\lfloor N/200 \rfloor$ węzłów
    \end{itemize}
    \vspace{0.2cm}
    \small Skalowanie proporcjonalne do $N$ zapobiega\\
    efektowi \textit{super-węzła} (setki agentów w 3 biurach).

    \column{0.48\textwidth}
    \begin{tabular}{lc}
      \toprule
      Typ węzła & $p_{\text{base}}$ \\
      \midrule
      Gospod.\ domowe & 0,35 \\
      Szkoła & 0,30 \\
      Biuro & 0,20 \\
      Placówka zdrowia & 0,15 \\
      Sklep & 0,10 \\
      Park & 0,03 \\
      \bottomrule
    \end{tabular}
    \vspace{0.3cm}

    \textbf{Dzienny harmonogram agenta:}\\
    Dom $\to$ Praca/Szkoła $\to$ Sklep? $\to$ Park?\\
    (zależny od \texttt{lockdown} i \texttt{social\_distancing})
  \end{columns}
\end{frame}

\begin{frame}{Mechanizmy transmisji}
  \textbf{Transmisja wewnątrz węzła POI (model przetrwania):}
  \[
    P_{\text{inf}} = \Biggl(1 - \prod_{i \in \text{contacts}}
      \bigl(1-p_{\text{eff},i}\bigr)^{v_i}\Biggr)
      \cdot (1 - \text{immunity})
  \]

  \textbf{Efektywne prawdopodobieństwo -- modyfikatory:}
  \[
    p_{\text{eff}} = p_{\text{base}} \cdot
    \begin{cases}
      (1-\eta_M)^2 & \text{obie strony w maseczce} \\
      (1-\eta_M)   & \text{jedna strona w maseczce} \\
      1            & \text{brak maseczek}
    \end{cases}
  \]
  \[
    h = 1 - 0{,}6 \cdot (\bar{\Theta}_H - 0{,}5),\qquad
    p_{\text{eff}}' = p_{\text{eff}} \cdot h \cdot (1 - 0{,}2 \cdot \mathbb{1}_{\text{SD}})
  \]

  \begin{alertblock}{Kluczowa poprawka: \texttt{MAX\_INFECTIOUS\_CONTACTS} = 5}
    Każdy podatny agent losuje \textbf{co najwyżej 5} spośród zakaźnych w węźle.
    Eliminuje artefaktualną saturację siły zakażenia w dużych węzłach.
  \end{alertblock}
\end{frame}

\begin{frame}{Atrybuty agentów}
  \begin{columns}[T]
    \column{0.50\textwidth}
    \textbf{Demograficzne:}
    \begin{itemize}
      \item \texttt{age} [0--80] -- mobilność, śmiertelność
      \item \texttt{household\_id} -- klaster domowy
      \item \texttt{workplace\_id} -- biuro lub szkoła
    \end{itemize}

    \vspace{0.25cm}
    \textbf{Kliniczne i immunologiczne:}
    \begin{itemize}
      \item \texttt{state} -- kompartment SEIRD
      \item \texttt{viral\_load} $\in$ [0,1] -- miano wirusa
      \item \texttt{immunity} $\in$ [0,1] -- odporność
      \item \texttt{vaccinated} -- redukcja transmisji i zgonów
    \end{itemize}

    \column{0.48\textwidth}
    \textbf{Behawioralne:}
    \begin{itemize}
      \item \texttt{wears\_mask} -- $p_{\text{eff}} \times (1-\eta_M)$
      \item \texttt{social\_distancing} -- $p_{\text{eff}} \times 0{,}8$, pominięcie sklepów
      \item \texttt{hygiene\_score} $\Theta_H$ -- czynnik $h \in [0{,}70;\;1{,}30]$
      \item \texttt{contact\_rate} -- skaluje $p_{\text{shop}}, p_{\text{park}}$
    \end{itemize}

    \vspace{0.3cm}
    \begin{block}{Profil superspreader}
      \texttt{contact\_rate} $= 3{,}0$\\
      \texttt{hygiene\_score} $= 0{,}1$
    \end{block}
  \end{columns}
\end{frame}

\section{Etap 1 -- Prototyp MVP}

\begin{frame}{Etap 1: Prototyp MVP na siatce}
  \begin{columns}[T]
    \column{0.46\textwidth}
    \textbf{Środowisko:} siatka $50\times50$, topologia torusowa

    \textbf{Populacja:} $N=500$, 5\% startowo zakaźnych

    \textbf{Ruch:} losowy (sąsiedztwo Moore'a), $p_{\text{inf}}=0{,}20$

    \vspace{0.3cm}
    \textbf{Wyznaczenie emergentnego $R_0$:}
    \[
      \bar{k} = 8 \cdot \underbrace{0{,}20}_{\rho} \cdot 0{,}95 \approx 1{,}52
    \]
    \[
      R_0 \approx p_{\text{inf}} \cdot \bar{k} \cdot T_I
             \approx 0{,}20 \cdot 1{,}52 \cdot 8 \approx \mathbf{2{,}4}
    \]

    \begin{block}{Weryfikacja}
      Stochastyczne $R_0 \approx 2{,}6$ -- zgodne z~[2,0; 3,5] dla wirusów oddechowych.
    \end{block}

    \column{0.52\textwidth}
    \includegraphics[width=\textwidth]{epidemic_curve.png}

    \vspace{0.1cm}
    {\small\textit{Krzywe SEIRD, MVP ($N=500$, 100 kroków). Widoczny szczyt I ok.\ dnia 25.}}
  \end{columns}
\end{frame}

\section{Etap 2 -- Implementacja POI}

\begin{frame}{Etap 2: Pełna implementacja środowiska POI}
  \begin{columns}[T]
    \column{0.50\textwidth}
    \textbf{Naprawione błędy kalibracyjne:}
    \begin{enumerate}
      \item \textbf{Super-węzeł:} skalowanie liczby węzłów z $N$ -- cel: 4 osoby/dom, 25/biuro
      \item \textbf{Mega-autobus:} tranzyt grupowany po \emph{konkretnym} węźle ID (nie po typie POI)
      \item \textbf{Saturacja:} \texttt{MAX\_INFECTIOUS\_CONTACTS = 5}
      \item \textbf{Kalibracja:} $p_{\text{mult}} = 0{,}16$ $\Rightarrow$ $R_0 \approx 2{,}5$
    \end{enumerate}

    \column{0.48\textwidth}
    \textbf{Stos technologiczny:}
    \begin{itemize}
      \item \textbf{Python 3.14} + \textbf{Mesa} (ABM framework)
      \item \textbf{NetworkX} -- graf POI
      \item \textbf{Streamlit} -- dashboard statyczny
      \item \textbf{HTML5 canvas} -- animacja real-time
      \item \textbf{pytest} -- 25 testów automatycznych
    \end{itemize}

    \vspace{0.25cm}
    \textbf{Testy obejmują:}\\
    przejścia E$\to$I$\to$R/D, blokady lockdown,\\
    właściwości \texttt{compute\_p\_eff}, wpływ szczepień.
  \end{columns}

  \vspace{0.4cm}
  \begin{alertblock}{Błąd o rząd wielkości}
    Pierwotny tranzyt: \emph{wszyscy} jadący do tego samego \textbf{typu} POI
    wchodzili do jednego pojazdu $\Rightarrow$ $R_0 \gg 10$. Naprawa: grupowanie po ID węzła.
  \end{alertblock}
\end{frame}

\begin{frame}{Etap 2: Analiza wrażliwości}
  \begin{columns}[T]

  \end{columns}

\section{Etap 3 -- Eksperymenty}

\begin{frame}{Etap 3: Metodologia eksperymentów}
  \begin{block}{Konfiguracja bazowa}
    $N=500$ agentów $\cdot$ 7 powtórzeń (różne ziarna) $\cdot$ 180 kroków\\
    Inkubacja 4 dni, zakaźność 8 dni, $p_{\text{death}}=2\%$, $p_{\text{mult}}=0{,}16$\\
    Startowe zakażenia: 2\% populacji w stanie~I
  \end{block}

  \vspace{0.4cm}
  \textbf{5 scenariuszy:}

  \begin{tabularx}{\textwidth}{@{}lX@{}}
    \toprule
    \textbf{Scenariusz} & \textbf{Zmiana vs.\ bazowy} \\
    \midrule
    1. Bazowy & Brak interwencji -- punkt odniesienia \\
    2. Lockdown & Zamknięcie biur, szkół, sklepów; agenci w domach \\
    3. Szczepienia 60\% & \texttt{immunity}=0,5; redukcja ryzyka zgonu o 80\% \\
    4. Superspreaderzy 5\% & CR=3,0; $\Theta_H=0{,}1$ dla 25 agentów \\
    5. Wysoka higiena & $\bar{\Theta}_H=0{,}85$ (bazowo 0,50) \\
    \bottomrule
  \end{tabularx}

  \vspace{0.3cm}
  Wykresy: \textbf{mediana} 7 powtórzeń + pasmo IQR (percentyle 25.--75.)

\begin{frame}{Scenariusze 1 \& 2: Baseline i Lockdown}
  \begin{columns}[T]
    \column{0.46\textwidth}

    \begin{itemize}
      \item Attack rate: \textbf{75,1\%}

    \begin{block}{Dlaczego tak skuteczny?}
      Lockdown przecina \emph{wszystkie} pozadomowe ścieżki transmisji.
      Pozostaje tylko przeniesienie wewnątrzdomowe o ograniczonym zasięgu.
    \end{block}

\section{Wnioski}

\begin{frame}{Wnioski końcowe}
  \begin{enumerate}
    \item \textbf{Lockdown jest najskuteczniejszą interwencją:}\\
          attack rate $75\% \to 3{,}5\%$ przez przecięcie wszystkich pozadomowych ścieżek transmisji.

    \vspace{0.3cm}
    \item \textbf{Szczepienia chronią głównie przed zgonem:}\\
          60\% pokrycia $\to$ $-35\%$ AR, $-55\%$ zgonów. Klaster w grupie niezaszczepionych omija odporność stadną.

    \vspace{0.3cm}
    \item \textbf{Superspreaderzy przyspieszają epidemię:}\\
          5\% populacji o CR=3, $\Theta_H=0{,}1$ skraca czas do szczytu o 1--2 dni (zasada 80/20).

    \vspace{0.3cm}
    \item \textbf{Higiena spłaszcza krzywą, ale nie eliminuje epidemii:}\\
          $\Delta\Theta_H = +0{,}35$ $\Rightarrow$ $-21$ p.p.\ AR, $+18$ dni szczytu.
          Niewystarczające samodzielnie przy $R_0 > 1$.
  \end{enumerate}

  \vspace{0.3cm}
  \begin{block}{Zaleta ABM wobec klasycznych ODE}
    Możliwość badania \textbf{heterogenicznych} efektów interwencji --
    wpływ rozkładów \texttt{contact\_rate} i \texttt{hygiene\_score}
    na makroskopową dynamikę epidemii.
  \end{block}
\end{frame}
